    /*
     * 1. DYNAMIC DATE DISPLAY
     *    Builds a formatted date string like "Wednesday, September 21st, 2022"
     *    and injects it into the banner's #date-display span.
     */
    (function displayDynamicDate() {
      var days   = ["Sunday","Monday","Tuesday","Wednesday","Thursday","Friday","Saturday"];
      var months = ["January","February","March","April","May","June",
                    "July","August","September","October","November","December"];

      var now     = new Date();
      var dayName = days[now.getDay()];
      var month   = months[now.getMonth()];
      var date    = now.getDate();
      var year    = now.getFullYear();

      /* Ordinal suffix helper: 1st, 2nd, 3rd, 4th … */
      function ordinal(n) {
        var s = ["th","st","nd","rd"];
        var v = n % 100;
        return n + (s[(v - 20) % 10] || s[v] || s[0]);
      }

      var dateString = dayName + ", " + month + " " + ordinal(date) + ", " + year;

      /* Write into the banner element */
      var el = document.getElementById("date-display");
      if (el) {
        el.textContent = "Today is: " + dateString;
      }

      /* Also update the footer copyright year */
      var yearEl = document.getElementById("footer-year");
      if (yearEl) {
        yearEl.textContent = year;
      }
    })();

    /*
     * 2. FORM VALIDATION
     *    Runs additional client-side checks that HTML5 patterns alone cannot
     *    cover — specifically: password confirmation match, and ensuring
     *    required radio button groups have a selection.
     */
    document.getElementById("registration-form").addEventListener("submit", function(event) {

      var valid = true;
      var messages = [];

      /* Check password confirmation */
      var pw1 = document.getElementById("password").value;
      var pw2 = document.getElementById("confirm_password").value;
      if (pw1 !== pw2) {
        valid = false;
        messages.push("Passwords do not match. Please re-enter your password.");
      }

      /* Check radio groups that are required (HTML5 'required' on radio
         works only when at least one in the group has required="required"
         and the browser supports it — we double-check here for safety.) */
      var radioGroups = ["gender", "vaccinated", "has_insurance", "smoker"];
      var radioLabels = {
        gender:        "Gender",
        vaccinated:    "Vaccination status",
        has_insurance: "Insurance status",
        smoker:        "Smoking status"
      };

      radioGroups.forEach(function(groupName) {
        var selected = document.querySelector("input[name='" + groupName + "']:checked");
        if (!selected) {
          valid = false;
          messages.push(radioLabels[groupName] + " must be selected.");
        }
      });

      /* If any check failed, prevent submission and alert the user */
      if (!valid) {
        event.preventDefault();
        alert("Please correct the following before submitting:\n\n" + messages.join("\n"));
      }
      /* If valid, the form's action="thankyou.html" will handle navigation */

    });
  </script>
